**2. New model: ExaminationQuestion (question library per credit module)**

```prisma
// ─── v7.0 Examination Question (account-level question library per credit module) ──

model ExaminationQuestion {
  id       String @id @default(dbgenerated("gen_random_uuid()")) @db.Uuid
  tenantId String @db.Uuid
  tenant   Tenant @relation(fields: [tenantId], references: [id], onDelete: Cascade)

  moduleCode     String   // ExaminationNode code (e.g., "CRD-HLN") — decoupled linkage
  text           String   @db.Text  // Question text
  rbiReference   String?  // RBI regulation area (e.g., "Master Direction on Housing Finance")
  bestPracticeTip String? @db.Text  // 1-2 sentence practical guidance
  category       String?  // Grouping within module (e.g., "Documentation", "Collateral", "NPA Norms")

  weight     Decimal @db.Decimal(5, 4) @default(1.0) // Weight for scoring
  isCritical Boolean @default(false) // Critical item: violation caps module score

  displayOrder Int     @default(0)  // For management UI ordering
  isActive     Boolean @default(true) // Soft-delete: false hides from new examinations

  createdById  String? @db.Uuid // NULL for system-seeded defaults
  createdAt    DateTime @default(now())
  updatedAt    DateTime @updatedAt

  // Relations
  accountExamResponses AccountExamResponse[]

  @@unique([tenantId, moduleCode, text])
  @@index([tenantId])
  @@index([tenantId, moduleCode, isActive])
  @@index([tenantId, moduleCode, displayOrder])
}
```

**3. New model: LoanAccount (uploaded loan portfolio per engagement)**

```prisma
// ─── v7.0 Loan Account (uploaded branch loan portfolio for sample-based examination) ──

model LoanAccount {
  id       String @id @default(dbgenerated("gen_random_uuid()")) @db.Uuid
  tenantId String @db.Uuid
  tenant   Tenant @relation(fields: [tenantId], references: [id], onDelete: Cascade)

  engagementId String          @db.Uuid
  engagement   AuditEngagement @relation(fields: [engagementId], references: [id], onDelete: Cascade)

  branchId   String @db.Uuid
  moduleCode String  // Credit module code (e.g., "CRD-HLN")

  // Core mandatory fields (same naming as LoanReview + sanctionDate)
  accountNo         String
  borrowerName      String
  productType       String         // "Housing Loan", "Gold Loan", "Vehicle Loan", etc.
  sanctionAmount    Decimal        @db.Decimal(15, 2)
  sanctionDate      DateTime       // For "newly sanctioned" sampling criteria
  outstandingAmount Decimal        @db.Decimal(15, 2)
  assetClass        String         // "STANDARD", "SMA0", "SMA1", "SMA2", "NPA_SUB", "NPA_DOUBTFUL", "NPA_LOSS"
  dpd               Int            @default(0) // Days Past Due

  // JSONB for module-specific fields (e.g., collateral type for Housing, gold purity for Gold Loans)
  metadata          Json?          // Flexible per-module fields

  // Sampling state
  isSampled         Boolean        @default(false)
  sampledAt         DateTime?

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  // Relations
  accountExamResponses AccountExamResponse[]

  @@unique([engagementId, accountNo])
  @@index([tenantId])
  @@index([engagementId])
  @@index([engagementId, moduleCode])
  @@index([engagementId, isSampled])
  @@index([engagementId, assetClass])
  @@index([engagementId, dpd])
}
```

**4. New model: SamplingConfig (HIA-configured sampling criteria per engagement/module)**

```prisma
// ─── v7.0 Sampling Config (HIA-defined sampling criteria per engagement/module) ──

model SamplingConfig {
  id       String @id @default(dbgenerated("gen_random_uuid()")) @db.Uuid
  tenantId String @db.Uuid
  tenant   Tenant @relation(fields: [tenantId], references: [id], onDelete: Cascade)

  engagementId String          @db.Uuid
  engagement   AuditEngagement @relation(fields: [engagementId], references: [id], onDelete: Cascade)

  moduleCode       String    // Credit module code (e.g., "CRD-HLN")
  sampleSizePct    Decimal   @db.Decimal(5, 2) // Overall sample size % (e.g., 10.00)

  // Criteria bucket allocations — JSONB array of { bucket, pct, description }
  // e.g., [{ "bucket": "NEWLY_SANCTIONED", "pct": 20, "description": "Loans sanctioned in last 12 months" }, ...]
  // Buckets: NEWLY_SANCTIONED, AMOUNT_WISE, AGE_WISE, DPD_WISE, PRIOR_OBSERVATIONS
  criteriaBuckets  Json      // Array of bucket allocations, must sum to 100

  isLocked         Boolean   @default(false)  // Once locked, auditors cannot modify
  lockedAt         DateTime?
  lockedById       String?   @db.Uuid

  // Snapshot of generated sample metadata
  sampleGenerated  Boolean   @default(false)
  sampleGeneratedAt DateTime?
  sampleCount      Int?      // Number of accounts selected

  createdById String  @db.Uuid
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt

  @@unique([engagementId, moduleCode])
  @@index([tenantId])
  @@index([engagementId])
}
```

```prisma
// ─── v7.0 Account Exam Response (per-account per-question examination response) ──

model AccountExamResponse {
  id       String @id @default(dbgenerated("gen_random_uuid()")) @db.Uuid
  tenantId String @db.Uuid

  engagementId String          @db.Uuid
  engagement   AuditEngagement @relation(fields: [engagementId], references: [id], onDelete: Cascade)

  loanAccountId String      @db.Uuid
  loanAccount   LoanAccount @relation(fields: [loanAccountId], references: [id], onDelete: Cascade)

  questionId String              @db.Uuid
  question   ExaminationQuestion @relation(fields: [questionId], references: [id])

  status     AccountExamResponseStatus  // COMPLIANT or VIOLATION
  note       String?  @db.Text          // Auditor's finding notes

  // Audit trail
  respondedById String   @db.Uuid
  respondedAt   DateTime @default(now())

  // Evidence
  evidence Evidence[] @relation("AccountExamResponseEvidence")

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  @@unique([engagementId, loanAccountId, questionId])
  @@index([tenantId])
  @@index([engagementId])
  @@index([engagementId, questionId])
  @@index([loanAccountId])
}
```
