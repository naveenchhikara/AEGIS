Create `src/data-access/__tests__/tenant-isolation.test.ts` — a Vitest integration test that verifies tenant data isolation.

**Important:** This test verifies DAL function patterns, not live DB queries (which would require a running database). It is a static analysis test that scans DAL source files for correct tenant isolation patterns.

```typescript
import { describe, it, expect } from "vitest";
import { readFileSync, readdirSync } from "fs";
import { join } from "path";

const DAL_DIR = join(process.cwd(), "src/data-access");

describe("Tenant Data Isolation (DSEC-05)", () => {
  // Get all .ts DAL files (excluding __tests__, index, types)
  const dalFiles = readdirSync(DAL_DIR)
    .filter((f) => f.endsWith(".ts") && !f.startsWith("__") && f !== "index.ts" && f !== "types.ts");

  it("all DAL files exist", () => {
    expect(dalFiles.length).toBeGreaterThan(0);
  });

  describe("every DAL file with findMany/findFirst includes tenantId filter", () => {
    for (const file of dalFiles) {
      const content = readFileSync(join(DAL_DIR, file), "utf-8");

      // Skip files that don't do any queries
      const hasQuery = /\.(findMany|findFirst|findUnique|count|aggregate)\b/.test(content);
      if (!hasQuery) continue;

      it(`${file} — queries include tenantId filter`, () => {
        // Check that tenantId appears in the file (as a WHERE clause parameter)
        const hasTenantFilter = content.includes("tenantId");
        expect(hasTenantFilter).toBe(true);
      });
    }
  });

  describe("no DAL function accepts tenantId from external input", () => {
    for (const file of dalFiles) {
      const content = readFileSync(join(DAL_DIR, file), "utf-8");

      it(`${file} — does not accept tenantId from request body/params`, () => {
        // tenantId should come from session, not from request body
        // Pattern: function parameters should not include { tenantId } from user input
        // This is a heuristic — check that tenantId is sourced from session
        const hasSessionImport = content.includes("getRequiredSession") ||
                                  content.includes("tenantId: string");
        // All DAL functions should either import session or accept tenantId as a typed parameter
        // (tenantId as string parameter is acceptable — it comes from session in the caller)
        expect(hasSessionImport).toBe(true);
      });
    }
  });

  it("prismaForTenant returns singleton (not tenant-scoped) — isolation is query-level", () => {
    const prismaFile = readFileSync(join(DAL_DIR, "prisma.ts"), "utf-8");
    // Verify the function exists and returns the global client
    expect(prismaFile).toContain("prismaForTenant");
    // Application-level isolation is documented and accepted
    // No RLS policies — WHERE clauses enforce isolation
  });
});
````

Adjust the exact assertions based on what you find when reading the actual DAL files. The key checks are:

2. tenantId comes from session context, not from user input
3. The prismaForTenant function exists in prisma.ts
